    // node-api/server.js
const express = require("express");
const { Pool } = require("pg");

const app = express();
const PORT = process.env.PORT || 3000;

const pool = new Pool({
  host: process.env.PGHOST || "postgres",
  port: Number(process.env.PGPORT || 5432),
  user: process.env.PGUSER || "demo",
  password: process.env.PGPASSWORD || "demo1234",
  database: process.env.PGDATABASE || "demo",
  connectionTimeoutMillis: 5000,
});

app.use(express.json());

app.get("/health", async (_req, res) => {
  try {
    const r = await pool.query("SELECT NOW() AS now");
    res.json({
      service: "node-api",
      status: "ok",
      runtime: `node ${process.version}`,
      database: "postgresql",
      dbTime: r.rows[0].now,
      time: new Date().toISOString(),
    });
  } catch (err) {
    res.status(503).json({ service: "node-api", status: "error", message: err.message });
  }
});

app.get("/messages", async (_req, res) => {
  try {
    const r = await pool.query(
      "SELECT id, name, content, created_at FROM messages ORDER BY id DESC LIMIT 100"
    );
    res.json(r.rows);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

app.post("/messages", async (req, res) => {
  const name = String(req.body.name || "").trim().slice(0, 40) || "anonymous";
  const content = String(req.body.content || "").trim().slice(0, 500);
  if (!content) {
    return res.status(400).json({ error: "content is required" });
  }
  try {
    const r = await pool.query(
      "INSERT INTO messages (name, content) VALUES ($1, $2) RETURNING id, name, content, created_at",
      [name, content]
    );
    res.status(201).json(r.rows[0]);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

app.delete("/messages/:id", async (req, res) => {
  const id = Number(req.params.id);
  if (!Number.isInteger(id)) return res.status(400).json({ error: "invalid id" });
  try {
    const r = await pool.query("DELETE FROM messages WHERE id = $1 RETURNING id", [id]);
    if (r.rowCount === 0) return res.status(404).json({ error: "not found" });
    res.json({ deleted: r.rows[0].id });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

app.listen(PORT, () => {
  console.log(`node-api listening on :${PORT}`);
});
    # python-api/app.py
import os
import platform
import socket

from flask import Flask, jsonify, request

app = Flask(__name__)
HOSTNAME = socket.gethostname()


@app.get("/health")
def health():
    return jsonify(
        service="python-api",
        status="ok",
        runtime=f"python {platform.python_version()}",
        host=HOSTNAME,
    )


@app.get("/info")
def info():
    return jsonify(
        service="python-api",
        python=platform.python_version(),
        implementation=platform.python_implementation(),
        platform=platform.platform(),
        host=HOSTNAME,
        cwd=os.getcwd(),
    )


@app.get("/fib")
def fib():
    try:
        n = int(request.args.get("n", "10"))
    except ValueError:
        return jsonify(error="n must be an integer"), 400
    if n < 0 or n > 90:
        return jsonify(error="n must be between 0 and 90"), 400
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return jsonify(n=n, result=a)


@app.get("/")
def index():
    return jsonify(service="python-api", endpoints=["/health", "/info", "/fib?n=10"])


if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.environ.get("PORT", "8000")))
    // rust-api/src/main.rs
use axum::{
    extract::Query,
    routing::get,
    Json, Router,
};
use serde::Serialize;
use serde_json::{json, Value};
use std::collections::HashMap;
use std::net::SocketAddr;

#[derive(Serialize)]
struct Health {
    service: &'static str,
    status: &'static str,
    runtime: &'static str,
}

async fn health() -> Json {
    Json(Health {
        service: "rust-api",
        status: "ok",
        runtime: "rust / axum",
    })
}

async fn fib(Query(params): Query>) -> Json {
    let n: u64 = params
        .get("n")
        .and_then(|v| v.parse().ok())
        .unwrap_or(20);

    if n > 90 {
        return Json(json!({ "error": "n must be <= 90" }));
    }

    let (mut a, mut b): (u128, u128) = (0, 1);
    for _ in 0..n {
        let next = a + b;
        a = b;
        b = next;
    }

    Json(json!({ "n": n, "result": a }))
}

async fn index() -> Json {
    Json(json!({
        "service": "rust-api",
        "endpoints": ["/health", "/fib?n=20"]
    }))
}

#[tokio::main]
async fn main() {
    let app = Router::new()
        .route("/", get(index))
        .route("/health", get(health))
        .route("/fib", get(fib));

    let addr = SocketAddr::from(([0, 0, 0, 0], 8002));
    let listener = tokio::net::TcpListener::bind(addr).await.expect("bind :8002");
    println!("rust-api listening on {addr}");
    axum::serve(listener, app).await.expect("server error");
}
    // frontend/app.js
const $ = (id) => document.getElementById(id);

async function getJSON(url, options) {
  const res = await fetch(url, options);
  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error(data.error || `${res.status} ${res.statusText}`);
  return data;
}

async function mark(el, url) {
  try {
    const data = await getJSON(url);
    el.classList.remove("down");
    el.classList.add("up");
    return data;
  } catch (err) {
    el.classList.remove("up");
    el.classList.add("down");
    return { error: err.message };
  }
}

async function checkStatus() {
  const results = await Promise.all([
    mark($("st-node"), "/api/health"),
    mark($("st-python"), "/py/health"),
    mark($("st-rust"), "/rust/health"),
  ]);
  $("status-json").textContent = JSON.stringify(
    { "node-api": results[0], "python-api": results[1], "rust-api": results[2] },
    null,
    2
  );
}

async function loadMessages() {
  try {
    const rows = await getJSON("/api/messages");
    $("msg-list").innerHTML = rows.length
      ? rows
          .map(
            (m) => `
  • 
              ${escapeHtml(m.name)}${new Date(
              m.created_at
            ).toLocaleString()}
              ${escapeHtml(m.content)}
              
            
    • `
          )
          .join("")
      : '
  • 尚無留言
    • ';

    document.querySelectorAll("#msg-list button").forEach((btn) => {
      btn.addEventListener("click", async () => {
        await getJSON(`/api/messages/${btn.dataset.id}`, { method: "DELETE" });
        loadMessages();
      });
    });
  } catch (err) {
    $("msg-list").innerHTML = `
  • 讀取失敗：${escapeHtml(
      err.message
    )}
    • `;
  }
}

function escapeHtml(s) {
  return String(s).replace(
    /[&<>"']/g,
    (c) => ({ "&": "&", "<": "<", ">": ">", '"': """, "'": "'" })[c]
  );
}

$("msg-form").addEventListener("submit", async (e) => {
  e.preventDefault();
  const name = $("name").value.trim();
  const content = $("content").value.trim();
  if (!content) return;
  try {
    await getJSON("/api/messages", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ name, content }),
    });
    $("content").value = "";
    loadMessages();
  } catch (err) {
    alert("送出失敗：" + err.message);
  }
});

$("btn-refresh").addEventListener("click", checkStatus);

async function loadService(url, el) {
  el.textContent = `loading ${url} …`;
  try {
    el.textContent = JSON.stringify(await getJSON(url), null, 2);
  } catch (err) {
    el.textContent = `error: ${err.message}`;
  }
}

$("btn-py").addEventListener("click", () => loadService("/py/info", $("svc-json")));
$("btn-rust").addEventListener("click", () =>
  loadService("/rust/fib?n=20", $("svc-json"))
);

checkStatus();
loadMessages();
setInterval(checkStatus, 15000);
